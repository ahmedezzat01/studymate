03The Solution — 6 Core Features

StudyMate unifies the entire study-group lifecycle — from discovery to rating — into a single intelligent platform.

🧠
Smart Academic Matching

Rule-based engine ranks groups by course overlap, schedule, study style & rating history.

📅
Session Scheduling

Calendar view with session reminders, time-slot coordination, and conflict detection.

💬
Real-Time Group Chat

Threaded conversations per group, file attachments, and inline reactions.

📁
Resource Sharing

Centralized library for PDFs, PPTXs, DOCXs, and links — searchable & version-controlled.

⭐
Session Rating System

After every session, members rate quality — drives the matching algorithm forward.

📊
Admin Dashboard

Group analytics, engagement metrics, and moderation tools for academic supervisors.

// lib/chat_screen.dart — Local state with TextEditingController
class _ChatScreenState extends State<ChatScreen> {
  final controller = TextEditingController();
  final localMessages = List<ChatMessage>.from(chatMessages);

  @override
  void dispose() {
    controller.dispose();        // Prevent memory leak
    super.dispose();
  }

  void sendMessage() {
    final text = controller.text.trim();
    if (text.isEmpty) return;
    setState(() {
      localMessages.add(ChatMessage(
        sender: 'You',
        text: text,
        time: 'Now',
        mine: true,
        avatar: 'SA',
        color: AppColors.primary,
      ));
      controller.clear();
    });
  }
}

// Interactive 5-star rating widget
class StarPicker extends StatefulWidget {
  final int initial;
  final ValueChanged<int> onChanged;
  const StarPicker({this.initial = 0, required this.onChanged});
  @override createState() => _StarPickerState();
}

class _StarPickerState extends State<StarPicker> {
  late int rating = widget.initial;
  @override
  Widget build(BuildContext ctx) => Row(
    mainAxisAlignment: MainAxisAlignment.center,
    children: List.generate(5, (i) => GestureDetector(
      onTap: () {
        setState(() => rating = i + 1);
        widget.onChanged(rating);
      },
      child: Padding(
        padding: const EdgeInsets.all(4),
        child: Icon(
          i < rating ? Icons.star_rounded : Icons.star_outline_rounded,
          size: 42,
          color: AppColors.orange,
        ),
      ),
    )),
  );
}

Type-safe data structures powering every screen. All models are const-constructible and immutable.

// lib/study_models.dart — 6 immutable data classes
class StudyGroup {
  final String name, course, description, reviews, time, place, members;
  final int match;
  final double rating;
  final Color color;
  final List<String> tags;
  const StudyGroup({required this.name, required this.course, ...});
}

class StudySession { final String course, title, time, place; final bool joined; }
class ChatMessage   { final String sender, text, time, avatar; final bool mine; final Color color; }
class SharedFile    { final String title, owner, details, type; final Color color; final IconData icon; }
class Notice        { final String title, body, time; final IconData icon; final Color color; final bool action; }
class GroupMember   { final String name, info, initials, label; final Color color; }
