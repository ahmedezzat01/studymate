Smart Matching

AI-Powered Academic Matching

A rule-based filtering engine that ranks groups by course overlap, schedule compatibility, study style, and rating history.

📥 Input

  • 📚 Courses Enrolled
  • 🕒 Schedule Slots
  • 🎯 Study Style
  • ⭐ Rating History
  • 🏫 University / Major

⚙️ Processing

  • 🔍 Course Overlap Filter
  • ⏰ Schedule Compatibility
  • 👥 Group Size & Activity
  • 📈 Rating Boost (≥ 4.0)
  • 🌐 Weighted Score (0–100%)

🎯 Output

  • Algorithm Avengers 92%
  • Calculus Crew 85%
  • Database Wizards 78%

// Mobile-first wrapper
class PhonePage extends StatelessWidget {
  final Widget child;
  const PhonePage({required this.child});
  @override
  Widget build(BuildContext ctx) =>
    Scaffold(
      body: Center(
        child: ConstrainedBox(
          constraints:
            const BoxConstraints(maxWidth: 390),
          child: child,
        ),
      ),
    );
}

// ChoiceChip filter system
final filters = ['Best Match','CS302','MATH201','CS410'];

  spacing: 8,
  children: filters.map((f) =>
    ChoiceChip(
      label: Text(f),
      selected: selected == f,
      onSelected: (_) => setState(() => selected = f),
    ),
  ).toList(),
)
chat_screen.dart
// Append & clear input
void sendMessage() {
  if (controller.text.trim().isEmpty) return;
  setState(() {
    messages.add({
      'text': controller.text,
      'mine': true,
      'time': TimeOfDay.now().format(ctx),
    });
    controller.clear();
  });
}
star_picker.dart
// Interactive star rating
Row(
  children: List.generate(5, (i) =>
    IconButton(
      icon: Icon(
        i < rating ? Icons.star : Icons.star_border,
        color: Colors.amber,
      ),
      onPressed: () => setState(() => rating = i + 1),
    ),
  ),
)
notifications_screen.dart
// Safe back navigation
void backToPreviousPage(BuildContext ctx) {
  if (Navigator.canPop(ctx)) {
    Navigator.pop(ctx);
  } else {
    Navigator.pushReplacementNamed(ctx, '/home');
  }
}
